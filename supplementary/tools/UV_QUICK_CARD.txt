┌─────────────────────────────────────────────────────────────────────────┐
│ PACKAGE INSTALLATION                                                    │
└─────────────────────────────────────────────────────────────────────────┘
  uv pip install requests                    # Single package
  uv pip install requests pandas numpy       # Multiple packages
  uv pip install requests==2.31.0            # Specific version
  uv pip install "requests>=2.28,<3.0"       # Version range
  uv pip install -r requirements.txt         # From file
  uv pip install -e .                        # Editable install
  uv pip install -e ".[dev]"                 # With extras

┌─────────────────────────────────────────────────────────────────────────┐
│ PACKAGE MANAGEMENT                                                      │
└─────────────────────────────────────────────────────────────────────────┘
  uv pip list                    # List all packages
  uv pip list --outdated         # Show outdated packages
  uv pip show requests           # Package details
  uv pip freeze                  # Show installed with versions
  uv pip freeze > requirements.txt          # Save to file
  uv pip uninstall requests                 # Remove package
  uv pip install --upgrade requests         # Upgrade package

┌─────────────────────────────────────────────────────────────────────────┐
│ REQUIREMENTS FILES                                                      │
└─────────────────────────────────────────────────────────────────────────┘
  uv pip compile requirements.in -o requirements.txt    # Lock deps
  uv pip sync requirements.txt                          # Exact sync
  uv pip install -r requirements.txt                    # Install

┌─────────────────────────────────────────────────────────────────────────┐
│ TROUBLESHOOTING                                                         │
└─────────────────────────────────────────────────────────────────────────┘
  ❌ uv not found
     → pip install uv --break-system-packages
     → export PATH="$HOME/.local/bin:$PATH"

  ❌ Can't activate venv
     → source .venv/bin/activate (not just "activate")
     → Check: ls -la .venv/

  ❌ Package conflicts
     → uv pip compile requirements.in -o requirements.txt
     → uv pip install -r requirements.txt
